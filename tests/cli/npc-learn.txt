# NPC teaching uses shared feature logic for eligibility, tuition, skills, and known state.
:new
:set char.money=100
:set skills=[]
:set room.npcs=[{"name":"师父","levelCap":1,"moveset":[{"id":"move_teaching_palm","name":"掌法","type":"攻击","archetype":"拳术","quality":"白"}]}]

:expect skills.length=1
:expect varTree.世界.已认识人物=["师父"]
