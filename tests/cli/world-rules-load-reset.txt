
# Loading an older save must initialize rules without ticking restored anger.
:new
:set time=8
:set varTree.角色.守卫.生气状态={"active":true,"turnsLeft":2}
:save world-rules-anger-two.json
打坐
:expect time=9
:expect varTree.角色.守卫.生气状态.turnsLeft=1
:load world-rules-anger-two.json
:expect time=8
:expect varTree.角色.守卫.生气状态.turnsLeft=2

:new
:set time=8
:set varTree.角色.守卫.生气状态={"active":true,"turnsLeft":1}
:save world-rules-anger-one.json
打坐
:expect time=9
:expect varTree.角色.守卫.生气状态.active=false
:load world-rules-anger-one.json
:expect time=8
:expect varTree.角色.守卫.生气状态.active=true
:expect varTree.角色.守卫.生气状态.turnsLeft=1
